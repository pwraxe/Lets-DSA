1  function generatePermute(arr, l, r) {
    if (l === r) {
        // Base case – permutation found
        results.push(arr.join(''));
        return;
    }
    for (let i = l; i <= r; i++) {
        // Swap current position with every remaining character
        [arr[l], arr[i]] = [arr[i], arr[l]];
        generatePermute(arr, l + 1, r);
        // Backtrack – restore for next iteration
        [arr[l], arr[i]] = [arr[i], arr[l]];
    }
}  // end of function
